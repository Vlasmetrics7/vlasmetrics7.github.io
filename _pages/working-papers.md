<script>
function toggleExtra(id) {
  const block = document.getElementById(id);
  if (!block) return;

  block.style.display = (block.style.display === "none" || block.style.display === "")
      ? "block"
      : "none";

  if (block.style.display === "block") {
    block.scrollIntoView({ behavior: "smooth", block: "start" });
  }
}

document.addEventListener("DOMContentLoaded", function () {
  var hashTargets = {
    'working-papers': 'workingPapersSection',
    'pop-science': 'popSection',
    'technical-notes': 'techSection',
    'book-articles': 'booksSection',
    'book': 'bookSection',
    'dissertations': 'dissertationSection'
  };
  var hash = window.location.hash.replace('#', '');
  var targetId = hashTargets[hash];
  if (targetId) {
    var el = document.getElementById(targetId);
    if (el) {
      el.style.display = 'block';
      el.scrollIntoView({ behavior: 'smooth', block: 'start' });
    }
  }
});
</script>
